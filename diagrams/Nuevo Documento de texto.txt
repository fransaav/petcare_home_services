C4Container
    title Diagrama de Contenedores - PetCare Home Services

    Person(customer, "Dueño de Mascota", "Cliente que reserva servicios para sus mascotas")
    Person(provider, "Proveedor de Servicios", "Profesional que atiende citas y actualiza el estado")

    System_Boundary(c1, "Sistema PetCare") {
        Container(api, "API REST", ".NET 10", "Punto de entrada unificado para clientes y proveedores")
        
        Container(mod_identity, "Módulo Identity & Pets", ".NET", "Gestiona perfiles de usuarios y mascotas")
        Container(mod_booking, "Módulo Booking", ".NET", "Gestiona las reservas y citas")
        Container(mod_providers, "Módulo Providers", ".NET", "Gestiona los perfiles y disponibilidad de proveedores")
        Container(mod_billing, "Módulo Billing", ".NET", "Gestiona pagos y facturación")
        
        ContainerDb(db, "SQL Server", "Base de Datos", "Almacena los datos. Cada módulo tiene su propio esquema/aislamiento")
        ContainerQueue(rabbitmq, "RabbitMQ", "Message Broker", "Bus de eventos para comunicación asíncrona")
    }

    Rel(customer, api, "Usa la aplicación / Hace reservas", "JSON/HTTPS")
    Rel(provider, api, "Usa la aplicación / Actualiza estados", "JSON/HTTPS")
    
    Rel(api, mod_identity, "Delega peticiones a")
    Rel(api, mod_booking, "Delega peticiones a")
    Rel(api, mod_providers, "Delega peticiones a")
    Rel(api, mod_billing, "Delega peticiones a")
    
    Rel(mod_identity, db, "Lee/Escribe (Esquema identity)", "EF Core")
    Rel(mod_booking, db, "Lee/Escribe (Esquema booking)", "EF Core")
    Rel(mod_providers, db, "Lee/Escribe (Esquema providers)", "EF Core")
    Rel(mod_billing, db, "Lee/Escribe (Esquema billing)", "EF Core")

    Rel(mod_booking, rabbitmq, "Publica BookingConfirmedEvent", "MassTransit")
    Rel(mod_providers, rabbitmq, "Publica ServiceStatusUpdatedEvent", "MassTransit")
    
    Rel(rabbitmq, mod_billing, "Consume BookingConfirmedEvent", "MassTransit")
    Rel(rabbitmq, mod_booking, "Consume ServiceStatusUpdatedEvent", "MassTransit")
